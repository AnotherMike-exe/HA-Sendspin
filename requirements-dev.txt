# --- Test harness ----------------------------------------------------------
#
# pytest-homeassistant-custom-component pins a specific Home Assistant version.
# Pin it deliberately and bump it on purpose — an unpinned bump silently changes
# which HA release the tests validate against.
#
# 0.13.355 pins homeassistant==2026.8.1, which is the live deploy target (see
# docs/DEPLOYMENT-TESTING.md). It requires Python 3.14 — phacc cut over to
# `requires_python >=3.14` at 0.13.317, so anything older silently drags the
# suite back to an HA release the integration is not shipped against.
#
# A bare `0.13.*` is NOT a pin: on Python 3.13 it resolves to 0.13.316, which
# pins homeassistant==2026.2.3 — six releases behind what we deploy to.
pytest-homeassistant-custom-component==0.13.355
ruff

# --- Runtime dependencies, mirrored ----------------------------------------
#
# These are declared for Home Assistant in custom_components/sendspin/manifest.json,
# which is what installs them on a real instance. They are duplicated here
# because the test suite imports them directly and nothing else would install
# them in CI.
#
# KEEP IN SYNC WITH manifest.json "requirements". The duplication is
# unavoidable — the same hazard as the zeroconf service types in const.py.
#
# numpy is not an optional extra: aiosendspin.server.server cannot be imported
# without it (the visualizer roles import numpy at module level, and they are
# imported eagerly for role registration). The artwork role imports PIL in the
# same way, but Pillow is a dependency of Home Assistant core. hassfest refuses a
# core dependency in a custom manifest, so Pillow is listed in neither file and
# arrives with `homeassistant`. `av` is deliberately excluded — it is only
# reached when encoding audio, which a source-less server never does.
aiosendspin==9.1.0
numpy>=1.26.0
